## Overview

### What This Benchmark Measures

1. **Emotion Recognition Accuracy**: Whether generated audio is correctly classified by emotion recognition models
2. **Acoustic Feature Correlation**: Whether prosodic features (pitch, energy, tempo) match expected emotion profiles
3. **Perceptual Quality**: Human evaluation of emotion expressiveness and naturalness
4. **Intensity Control**: Whether emotion intensity parameter produces proportional acoustic changes

## Emotion Validation Methods

### Method 1: Pre-trained Emotion Classifier

Use a pre-trained speech emotion recognition (SER) model to classify generated audio.

```python
#!/usr/bin/env python3
"""
evaluate_emotion_classifier.py

Validate generated audio using pre-trained emotion classifier.
Uses Hugging Face's emotion recognition models.
"""

import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent / "src"))

import torch
import torchaudio
import numpy as np
from transformers import AutoFeatureExtractor, AutoModelForAudioClassification
from collections import defaultdict
import json

# Pre-trained emotion recognition models
EMOTION_MODELS = {
    "wav2vec2-emotion": "ehcalabres/wav2vec2-lg-xlsr-en-speech-emotion-recognition",
    "hubert-emotion": "superb/hubert-large-superb-er",
}

def load_emotion_classifier(model_name="wav2vec2-emotion"):
    """Load pre-trained emotion classifier."""
    model_id = EMOTION_MODELS.get(model_name, model_name)
    feature_extractor = AutoFeatureExtractor.from_pretrained(model_id)
    model = AutoModelForAudioClassification.from_pretrained(model_id)
    return feature_extractor, model

def classify_emotion(audio_path, feature_extractor, model, target_sr=16000):
    """Classify emotion from audio file."""
    # Load audio
    waveform, sr = torchaudio.load(audio_path)

    # Resample if needed
    if sr != target_sr:
        resampler = torchaudio.transforms.Resample(sr, target_sr)
        waveform = resampler(waveform)

    # Convert to mono
    if waveform.shape[0] > 1:
        waveform = waveform.mean(dim=0, keepdim=True)

    # Extract features
    inputs = feature_extractor(
        waveform.squeeze().numpy(),
        sampling_rate=target_sr,
        return_tensors="pt"
    )

    # Classify
    with torch.no_grad():
        outputs = model(**inputs)
        logits = outputs.logits
        probs = torch.softmax(logits, dim=-1)

    # Get prediction
    pred_id = torch.argmax(probs, dim=-1).item()
    pred_label = model.config.id2label[pred_id]
    confidence = probs[0, pred_id].item()

    # Get all probabilities
    all_probs = {model.config.id2label[i]: probs[0, i].item()
                 for i in range(len(model.config.id2label))}

    return {
        "predicted": pred_label,
        "confidence": confidence,
        "all_probs": all_probs
    }

def evaluate_emotion_accuracy(audio_dir, expected_emotions, model_name="wav2vec2-emotion"):
    """
    Evaluate emotion classification accuracy for generated audio.

    Args:
        audio_dir: Directory containing generated audio files
        expected_emotions: Dict mapping filename patterns to expected emotions
        model_name: Name of the emotion classifier to use

    Returns:
        Dict with accuracy metrics
    """
    feature_extractor, model = load_emotion_classifier(model_name)

    results = []
    correct = 0
    total = 0

    audio_dir = Path(audio_dir)
    for audio_file in audio_dir.glob("*.wav"):
        # Find expected emotion from filename
        expected = None
        for pattern, emotion in expected_emotions.items():
            if pattern in audio_file.name:
                expected = emotion
                break

        if expected is None:
            continue

        # Classify
        result = classify_emotion(audio_file, feature_extractor, model)
        result["file"] = audio_file.name
        result["expected"] = expected
        result["correct"] = result["predicted"].lower() == expected.lower()

        results.append(result)
        total += 1
        if result["correct"]:
            correct += 1

        print(f"File: {audio_file.name}")
        print(f"  Expected: {expected}, Predicted: {result['predicted']} ({result['confidence']:.2%})")
        print(f"  Correct: {'Yes' if result['correct'] else 'No'}")

    accuracy = correct / total if total > 0 else 0

    return {
        "accuracy": accuracy,
        "correct": correct,
        "total": total,
        "results": results
    }

if __name__ == "__main__":
    # Example usage
    expected_emotions = {
        "emotion_happy": "happy",
        "emotion_sad": "sad",
        "emotion_angry": "angry",
        "emotion_neutral": "neutral",
        "emotion_fearful": "fear",
        "emotion_surprised": "surprise",
        "emotion_disgusted": "disgust",
        "emotion_calm": "calm",
    }

    results = evaluate_emotion_accuracy(".", expected_emotions)
    print(f"\nOverall Accuracy: {results['accuracy']:.2%} ({results['correct']}/{results['total']})")
```

### Method 2: Acoustic Feature Analysis (openSMILE)

Extract and analyze acoustic features that correlate with emotions.

```python
#!/usr/bin/env python3
"""
analyze_acoustic_features.py

Extract and analyze acoustic features using openSMILE.
"""

import opensmile
import pandas as pd
import numpy as np
from pathlib import Path
from sklearn.metrics import accuracy_score
from sklearn.ensemble import RandomForestClassifier
import json

# Expected acoustic profiles for each emotion
# Based on research: Scherer, K. R. (2003). Vocal communication of emotion
EMOTION_PROFILES = {
    "happy": {
        "f0_mean": "high",      # High pitch
        "f0_stddev": "high",    # High pitch variation
        "loudness_mean": "high", # Loud
        "tempo": "fast",        # Fast speaking rate
    },
    "sad": {
        "f0_mean": "low",       # Low pitch

    },

    },

        "loudness_mean": "medium", # Medium loudness
        "tempo": "medium",      # Medium speaking rate
    },
    "calm": {
        "f0_mean": "low",       # Low pitch
        "f0_stddev": "low",     # Low pitch variation
        "loudness_mean": "low",  # Soft
        "tempo": "slow",        # Slow speaking rate
    },
    "surprised": {
        "f0_mean": "high",      # High pitch
        "f0_stddev": "high",    # High pitch variation
        "loudness_mean": "high", # Loud
        "tempo": "medium",      # Medium speaking rate
    },
    "disgusted": {
        "f0_mean": "low",       # Low pitch
        "f0_stddev": "medium",  # Medium pitch variation
        "loudness_mean": "medium", # Medium loudness
        "tempo": "medium",      # Medium speaking rate
    },
}

def extract_opensmile_features(audio_path, feature_set="eGeMAPSv02"):
    """Extract acoustic features using openSMILE."""
    smile = opensmile.Smile(
        feature_set=opensmile.FeatureSet.eGeMAPSv02,
        feature_level=opensmile.FeatureLevel.Functionals,
    )

    features = smile.process_file(str(audio_path))
    return features

def analyze_emotion_features(audio_dir, pattern="merged_emotion_*.wav"):
    """
    Analyze acoustic features of generated audio files.

    Returns:
        DataFrame with features for each audio file
    """
    audio_dir = Path(audio_dir)
    all_features = []

    for audio_file in sorted(audio_dir.glob(pattern)):
        # Extract emotion from filename
        # Format: merged_emotion_happy.wav -> happy
        parts = audio_file.stem.split("_")
        if len(parts) >= 3:
            emotion = parts[2]
        else:
            emotion = "unknown"

        try:
            features = extract_opensmile_features(audio_file)
            features["emotion"] = emotion
            features["file"] = audio_file.name
            all_features.append(features)
            print(f"Extracted features from: {audio_file.name}")
        except Exception as e:
            print(f"Error processing {audio_file.name}: {e}")

    if all_features:
        df = pd.concat(all_features, ignore_index=True)
        return df
    return pd.DataFrame()

def compute_emotion_similarity(features_df):
    """
    Compute similarity between generated audio features and expected profiles.
    """
    # Key features to analyze
    key_features = [
        "F0semitoneFrom27.5Hz_sma3nz_amean",  # Mean pitch
        "F0semitoneFrom27.5Hz_sma3nz_stddevNorm",  # Pitch variation
        "loudness_sma3_amean",  # Mean loudness
        "loudness_sma3_stddevNorm",  # Loudness variation
    ]

    results = []

    for _, row in features_df.iterrows():
        emotion = row["emotion"]
        if emotion not in EMOTION_PROFILES:
            continue

        profile = EMOTION_PROFILES[emotion]

        # Compute percentile ranks for each feature
        # This helps compare relative feature values
        result = {
            "file": row["file"],
            "emotion": emotion,
        }

        for feature in key_features:
            if feature in features_df.columns:
                # Get percentile of this value within all samples
                all_values = features_df[feature].values
                value = row[feature]
                percentile = (all_values < value).sum() / len(all_values) * 100
                result[feature + "_percentile"] = percentile

        results.append(result)

    return pd.DataFrame(results)

def generate_feature_report(audio_dir, output_path="acoustic_analysis.json"):
    """Generate comprehensive acoustic feature report."""
    print("Extracting acoustic features...")
    features_df = analyze_emotion_features(audio_dir)

    if features_df.empty:
        print("No features extracted!")
        return

    print("\nComputing emotion similarity...")
    similarity_df = compute_emotion_similarity(features_df)

    # Generate report
    report = {
        "summary": {
            "total_files": len(features_df),
            "emotions_analyzed": features_df["emotion"].unique().tolist(),
        },
        "features_by_emotion": {},
        "similarity_analysis": similarity_df.to_dict(orient="records"),
    }

    # Compute mean features per emotion
    emotion_groups = features_df.groupby("emotion")
    key_features = [
        "F0semitoneFrom27.5Hz_sma3nz_amean",
        "F0semitoneFrom27.5Hz_sma3nz_stddevNorm",
        "loudness_sma3_amean",
    ]

    for emotion, group in emotion_groups:
        report["features_by_emotion"][emotion] = {}
        for feature in key_features:
            if feature in group.columns:
                report["features_by_emotion"][emotion][feature] = {
                    "mean": float(group[feature].mean()),
                    "std": float(group[feature].std()),
                }

    # Save report
    with open(output_path, "w") as f:
        json.dump(report, f, indent=2)
    print(f"\nReport saved to: {output_path}")

    # Print summary
    print("\n" + "=" * 60)
    print("ACOUSTIC FEATURE SUMMARY BY EMOTION")
    print("=" * 60)

    for emotion, features in report["features_by_emotion"].items():
        print(f"\n{emotion.upper()}:")
        for feature, stats in features.items():
            short_name = feature.split("_")[0]
            print(f"  {short_name}: {stats['mean']:.2f} (std: {stats['std']:.2f})")

    return report

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--audio_dir", type=str, default=".",
                       help="Directory containing generated audio files")
    parser.add_argument("--output", type=str, default="acoustic_analysis.json",
                       help="Output path for analysis report")
    args = parser.parse_args()

    generate_feature_report(args.audio_dir, args.output)
```

### Method 3: Prosodic Feature Comparison

Compare pitch, energy, and duration patterns.

```python
#!/usr/bin/env python3
"""
analyze_prosody.py

Analyze prosodic features (pitch, energy, duration) for emotion validation.
"""

import librosa
import numpy as np
import matplotlib.pyplot as plt
from pathlib import Path
import json

def extract_prosodic_features(audio_path, sr=24000):
    """
    Extract prosodic features from audio file.

    Returns:
        Dict with pitch, energy, and duration features
    """
    # Load audio
    y, sr = librosa.load(audio_path, sr=sr)

    # Duration
    duration = len(y) / sr

    # Pitch (F0) using PYIN
    f0, voiced_flag, voiced_probs = librosa.pyin(
        y, fmin=50, fmax=500, sr=sr
    )
    f0_clean = f0[~np.isnan(f0)]

    # Energy (RMS)
    rms = librosa.feature.rms(y=y)[0]

    # Spectral features
    spectral_centroid = librosa.feature.spectral_centroid(y=y, sr=sr)[0]
    spectral_rolloff = librosa.feature.spectral_rolloff(y=y, sr=sr)[0]

    # Zero crossing rate (correlates with pitch perception)
    zcr = librosa.feature.zero_crossing_rate(y)[0]

    # Tempo
    tempo, _ = librosa.beat.beat_track(y=y, sr=sr)

    features = {
        "duration_s": duration,
        "pitch": {
            "mean_hz": float(np.mean(f0_clean)) if len(f0_clean) > 0 else 0,
            "std_hz": float(np.std(f0_clean)) if len(f0_clean) > 0 else 0,
            "min_hz": float(np.min(f0_clean)) if len(f0_clean) > 0 else 0,
            "max_hz": float(np.max(f0_clean)) if len(f0_clean) > 0 else 0,
            "range_hz": float(np.max(f0_clean) - np.min(f0_clean)) if len(f0_clean) > 0 else 0,
            "voiced_fraction": float(np.mean(voiced_flag)),
        },
        "energy": {
            "mean": float(np.mean(rms)),
            "std": float(np.std(rms)),
            "max": float(np.max(rms)),
            "dynamic_range_db": float(20 * np.log10(np.max(rms) / (np.min(rms) + 1e-10))),
        },
        "spectral": {
            "centroid_mean": float(np.mean(spectral_centroid)),
            "centroid_std": float(np.std(spectral_centroid)),
            "rolloff_mean": float(np.mean(spectral_rolloff)),
        },
        "tempo_bpm": float(tempo) if isinstance(tempo, (int, float)) else float(tempo[0]) if len(tempo) > 0 else 0,
        "zcr_mean": float(np.mean(zcr)),
    }

    return features

def analyze_emotion_prosody(audio_dir, pattern="merged_emotion_*.wav"):
    """
    Analyze prosodic features for all generated emotion audio.
    """
    audio_dir = Path(audio_dir)
    results = {}

    for audio_file in sorted(audio_dir.glob(pattern)):
        # Extract emotion from filename
        parts = audio_file.stem.split("_")
        if len(parts) >= 3:
            emotion = parts[2]
        else:
            emotion = "unknown"

        print(f"Analyzing: {audio_file.name}...")
        features = extract_prosodic_features(audio_file)
        features["file"] = audio_file.name
        results[emotion] = features

    return results

def plot_prosody_comparison(results, output_path="prosody_comparison.png"):
    """
    Create visualization comparing prosodic features across emotions.
    """
    emotions = list(results.keys())

    # Extract features for plotting
    pitch_means = [results[e]["pitch"]["mean_hz"] for e in emotions]
    pitch_stds = [results[e]["pitch"]["std_hz"] for e in emotions]
    energy_means = [results[e]["energy"]["mean"] for e in emotions]
    tempos = [results[e]["tempo_bpm"] for e in emotions]

    fig, axes = plt.subplots(2, 2, figsize=(12, 10))

    # Pitch mean
    ax1 = axes[0, 0]
    bars1 = ax1.bar(emotions, pitch_means, color='steelblue')
    ax1.set_ylabel('Hz')
    ax1.set_title('Mean Pitch (F0)')
    ax1.tick_params(axis='x', rotation=45)

    # Pitch variation
    ax2 = axes[0, 1]
    bars2 = ax2.bar(emotions, pitch_stds, color='coral')
    ax2.set_ylabel('Hz')
    ax2.set_title('Pitch Variation (Std)')
    ax2.tick_params(axis='x', rotation=45)

    # Energy
    ax3 = axes[1, 0]
    bars3 = ax3.bar(emotions, energy_means, color='green')
    ax3.set_ylabel('RMS')
    ax3.set_title('Mean Energy')
    ax3.tick_params(axis='x', rotation=45)

    # Tempo
    ax4 = axes[1, 1]
    bars4 = ax4.bar(emotions, tempos, color='purple')
    ax4.set_ylabel('BPM')
    ax4.set_title('Tempo')
    ax4.tick_params(axis='x', rotation=45)

    plt.tight_layout()
    plt.savefig(output_path, dpi=150)
    print(f"\nPlot saved to: {output_path}")
    plt.close()

def generate_prosody_report(audio_dir, output_path="prosody_report.json"):
    """Generate comprehensive prosody analysis report."""
    results = analyze_emotion_prosody(audio_dir)

    # Save JSON report
    with open(output_path, "w") as f:
        json.dump(results, f, indent=2)
    print(f"\nReport saved to: {output_path}")

    # Create visualization
    plot_prosody_comparison(results)

    # Print summary
    print("\n" + "=" * 60)
    print("PROSODIC FEATURE SUMMARY BY EMOTION")
    print("=" * 60)

    for emotion, features in results.items():
        print(f"\n{emotion.upper()}:")
        print(f"  Pitch: {features['pitch']['mean_hz']:.1f} Hz (std: {features['pitch']['std_hz']:.1f})")
        print(f"  Energy: {features['energy']['mean']:.4f} (range: {features['energy']['dynamic_range_db']:.1f} dB)")
        print(f"  Tempo: {features['tempo_bpm']:.1f} BPM")

    return results

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--audio_dir", type=str, default=".",
                       help="Directory containing generated audio files")
    parser.add_argument("--output", type=str, default="prosody_report.json",
                       help="Output path for prosody report")
    args = parser.parse_args()

    generate_prosody_report(args.audio_dir, args.output)
```

```python
def run_benchmark(
    checkpoint_path: str,
    output_dir: str,
    device: str = "auto",
    audio_prompt: str = None,
):
    """
    Run complete emotion benchmark.

    Args:
        checkpoint_path: Path to emotion checkpoint
        output_dir: Directory to save generated audio
        device: Device to use
        audio_prompt: Optional reference audio for voice cloning
    """
    # Setup device
    if device == "auto":
        if torch.cuda.is_available():
            device = "cuda"
        elif torch.backends.mps.is_available():
            device = "cpu"  # MPS has issues
        else:
            device = "cpu"

    print(f"Using device: {device}")

    # Create output directory
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    # Load model
    print("Loading model...")
    model = ChatterboxMultilingualTTS.from_pretrained(device=device)

    # Load emotion checkpoint
    if checkpoint_path and Path(checkpoint_path).exists():
        print(f"Loading emotion checkpoint: {checkpoint_path}")
        model.load_emotion_checkpoint(checkpoint_path)
    else:
        print("Warning: No checkpoint loaded, using base model")

    # Results
    results = {
        "timestamp": datetime.now().isoformat(),
        "checkpoint": checkpoint_path,
        "device": device,
        "audio_prompt": audio_prompt,
        "test_cases": [],
    }

    # Run basic emotion tests
    print("\n" + "=" * 60)
    print("BASIC EMOTION TESTS")
    print("=" * 60)

    for text, emotion, intensity, desc in BENCHMARK_CASES["basic_emotions"]:
        print(f"\nGenerating: {emotion} ({desc})")
        print(f"  Text: {text}")

        try:
            wav = model.generate(
                text=text,
                language_id="en",
                audio_prompt_path=audio_prompt,
                emotion=emotion,
                emotion_intensity=intensity,
                temperature=0.8,
                cfg_weight=0.5,
            )

            filename = f"benchmark_{emotion}_{desc}.wav"
            filepath = output_dir / filename

            # Save audio
            if isinstance(wav, torch.Tensor):
                wav_np = wav.cpu().numpy().squeeze()
            else:
                wav_np = wav.squeeze()
            sf.write(filepath, wav_np, model.sr)

            duration = len(wav_np) / model.sr
            print(f"  Saved: {filename} ({duration:.2f}s)")

            results["test_cases"].append({
                "type": "basic",
                "text": text,
                "emotion": emotion,
                "intensity": intensity,
                "description": desc,
                "file": filename,
                "duration": duration,
                "status": "success",
            })

        except Exception as e:
            print(f"  Error: {e}")
            results["test_cases"].append({
                "type": "basic",
                "text": text,
                "emotion": emotion,
                "intensity": intensity,
                "description": desc,
                "status": "error",
                "error": str(e),
            })

    # Run intensity tests
    print("\n" + "=" * 60)
    print("INTENSITY VARIATION TESTS")
    print("=" * 60)

    for text, emotion, intensity, desc in BENCHMARK_CASES["intensity_test"]:
        print(f"\nGenerating: {emotion} at intensity {intensity}")

        try:
            wav = model.generate(
                text=text,
                language_id="en",
                audio_prompt_path=audio_prompt,
                emotion=emotion,
                emotion_intensity=intensity,
                temperature=0.8,
                cfg_weight=0.5,
            )

            filename = f"benchmark_intensity_{emotion}_{intensity:.1f}.wav"
            filepath = output_dir / filename

            if isinstance(wav, torch.Tensor):
                wav_np = wav.cpu().numpy().squeeze()
            else:
                wav_np = wav.squeeze()
            sf.write(filepath, wav_np, model.sr)

            duration = len(wav_np) / model.sr
            print(f"  Saved: {filename} ({duration:.2f}s)")

            results["test_cases"].append({
                "type": "intensity",
                "text": text,
                "emotion": emotion,
                "intensity": intensity,
                "description": desc,
                "file": filename,
                "duration": duration,
                "status": "success",
            })

        except Exception as e:
            print(f"  Error: {e}")
            results["test_cases"].append({
                "type": "intensity",
                "text": text,
                "emotion": emotion,
                "intensity": intensity,
                "description": desc,
                "status": "error",
                "error": str(e),
            })

    # Run blend tests
    print("\n" + "=" * 60)
    print("EMOTION BLENDING TESTS")
    print("=" * 60)

    for text, blend, desc in BENCHMARK_CASES["blend_test"]:
        print(f"\nGenerating blend: {desc}")
        print(f"  Weights: {blend}")

        try:
            wav = model.generate(
                text=text,
                language_id="en",
                audio_prompt_path=audio_prompt,
                emotion_blend=blend,
                temperature=0.8,
                cfg_weight=0.5,
            )

            filename = f"benchmark_blend_{desc}.wav"
            filepath = output_dir / filename

            if isinstance(wav, torch.Tensor):
                wav_np = wav.cpu().numpy().squeeze()
            else:
                wav_np = wav.squeeze()
            sf.write(filepath, wav_np, model.sr)

            duration = len(wav_np) / model.sr
            print(f"  Saved: {filename} ({duration:.2f}s)")

            results["test_cases"].append({
                "type": "blend",
                "text": text,
                "emotion_blend": blend,
                "description": desc,
                "file": filename,
                "duration": duration,
                "status": "success",
            })

        except Exception as e:
            print(f"  Error: {e}")
            results["test_cases"].append({
                "type": "blend",
                "text": text,
                "emotion_blend": blend,
                "description": desc,
                "status": "error",
                "error": str(e),
            })

    # Save results
    results_path = output_dir / "benchmark_results.json"
    with open(results_path, "w") as f:
        json.dump(results, f, indent=2)
    print(f"\n\nResults saved to: {results_path}")

    # Print summary
    successful = sum(1 for tc in results["test_cases"] if tc["status"] == "success")
    total = len(results["test_cases"])
    print(f"\nBenchmark complete: {successful}/{total} tests successful")

    return results

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Run emotion TTS benchmark")
```

## Human Evaluation Protocol

### Listening Test Setup

```python
#!/usr/bin/env python3
"""
human_eval_setup.py

Setup for human evaluation listening tests.
"""

import random
import json
from pathlib import Path
import shutil

def prepare_listening_test(benchmark_dir, output_dir, n_samples=20):
    """
    Prepare randomized listening test materials.

    Creates:
    - Randomized audio files (anonymized names)
    - Answer key (for evaluation)
    - Rating form template
    """
    benchmark_dir = Path(benchmark_dir)
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    # Collect all audio files
    audio_files = list(benchmark_dir.glob("benchmark_*.wav"))

    # Sample if needed
    if n_samples and n_samples < len(audio_files):
        audio_files = random.sample(audio_files, n_samples)

    # Shuffle
    random.shuffle(audio_files)

    # Create anonymized copies and answer key
    answer_key = []

    for i, src_file in enumerate(audio_files):
        # Create anonymized name
        anon_name = f"sample_{i+1:03d}.wav"
        dst_path = output_dir / anon_name

        # Copy file
        shutil.copy(src_file, dst_path)

        # Parse original emotion from filename
        parts = src_file.stem.split("_")
        if "blend" in parts:
            emotion = "blend"
        elif "intensity" in parts:
            emotion = parts[2]
            intensity = parts[3]
        else:
            emotion = parts[1]

        answer_key.append({
            "sample_id": i + 1,
            "anonymous_file": anon_name,
            "original_file": src_file.name,
            "expected_emotion": emotion,
        })

    # Save answer key
    with open(output_dir / "answer_key.json", "w") as f:
        json.dump(answer_key, f, indent=2)

    # Create rating form template
    rating_form = {
        "instructions": """
EMOTION RECOGNITION LISTENING TEST

For each audio sample, please:
1. Listen to the audio carefully (you may replay up to 3 times)
2. Select the emotion you perceive from the list below
3. Rate the emotion intensity (1=subtle, 5=very strong)
4. Rate the overall naturalness (1=robotic, 5=very natural)

Emotions: neutral, happy, sad, angry, fearful, calm, surprised, disgusted, excited, mixed

        """,
        "samples": []
    }

    for i in range(len(audio_files)):
        rating_form["samples"].append({
            "sample_id": i + 1,
            "file": f"sample_{i+1:03d}.wav",
            "perceived_emotion": "",
            "emotion_intensity": 0,  # 1-5
            "naturalness": 0,  # 1-5
            "comments": ""
        })

    with open(output_dir / "rating_form.json", "w") as f:
        json.dump(rating_form, f, indent=2)

    print(f"Listening test prepared in: {output_dir}")
    print(f"  - {len(audio_files)} audio samples")
    print(f"  - answer_key.json (for evaluator)")
    print(f"  - rating_form.json (for participants)")

    return answer_key

def compute_human_eval_metrics(answer_key_path, ratings_paths):
    """
    Compute metrics from human evaluation ratings.

    Args:
        answer_key_path: Path to answer_key.json
        ratings_paths: List of paths to completed rating forms
    """
    # Load answer key
    with open(answer_key_path) as f:
        answer_key = json.load(f)

    # Load all ratings
    all_ratings = []
    for path in ratings_paths:
        with open(path) as f:
            all_ratings.append(json.load(f))

    # Compute metrics
    metrics = {
        "n_raters": len(all_ratings),
        "n_samples": len(answer_key),
        "emotion_accuracy": 0.0,
        "avg_intensity": 0.0,
        "avg_naturalness": 0.0,
        "per_emotion": {},
    }

    # Map sample IDs to expected emotions
    expected = {item["sample_id"]: item["expected_emotion"]
                for item in answer_key}

    correct = 0
    total = 0
    intensity_sum = 0
    naturalness_sum = 0

    for ratings in all_ratings:
        for sample in ratings["samples"]:
            sample_id = sample["sample_id"]
            perceived = sample["perceived_emotion"].lower()
            expected_emotion = expected[sample_id].lower()

            if perceived == expected_emotion:
                correct += 1
            total += 1

            intensity_sum += sample["emotion_intensity"]
            naturalness_sum += sample["naturalness"]

    metrics["emotion_accuracy"] = correct / total if total > 0 else 0
    metrics["avg_intensity"] = intensity_sum / total if total > 0 else 0
    metrics["avg_naturalness"] = naturalness_sum / total if total > 0 else 0

    return metrics

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--benchmark_dir", type=str, default="benchmark_output")
    parser.add_argument("--output_dir", type=str, default="listening_test")
    parser.add_argument("--n_samples", type=int, default=20)
    args = parser.parse_args()

    prepare_listening_test(args.benchmark_dir, args.output_dir, args.n_samples)
```
